"""Q2 정답 예시 코드(교사용): 서버 로그 수집기를 SRP·DIP에 맞게 나눈 한 가지 구조 (Python 3.12+)

Q2는 분석 문항이라 학생이 코드를 제출하지 않는다. 이 파일은 (3) 서술형 '개선 구조'의
모범 예시이며, 시작 코드와 결과가 같음을 assert로 확인한다.

실행: python3 q2_log_collector.py   -> 마지막 줄 "Q2 정답 코드: 모든 assert 통과"
"""
import os
import tempfile
from abc import ABC, abstractmethod


class LogParser:
    """변경 이유: 로그 형식(수집 에이전트 팀)."""
    def parse(self, raw_text):
        entries = []
        for row in raw_text.strip().splitlines():
            level, server, message = row.split("|")
            entries.append((level.strip(), server.strip(), message.strip()))
        return entries


class LogStorage(ABC):
    """저장 위치에 대한 추상화. 상위 정책(LogPipeline)은 이것만 안다."""
    @abstractmethod
    def write(self, lines): ...


class FileLogStorage(LogStorage):
    """변경 이유: 저장 위치(인프라 팀). 파일 열기는 여기에만 있다."""
    def __init__(self, path):
        self.path = path

    def write(self, lines):
        with open(self.path, "w", encoding="utf-8") as f:
            for line in lines:
                f.write(line + "\n")


class MemoryLogStorage(LogStorage):
    """테스트용 저장소: 파일 없이 검증할 수 있다(DIP의 이점)."""
    def __init__(self):
        self.lines = []

    def write(self, lines):
        self.lines = list(lines)


class SummaryFormatter:
    """변경 이유: 회의용 요약 문장 형식(운영팀)."""
    def format(self, errors):
        servers = sorted({server for _, server, _ in errors})
        return f"오류 {len(errors)}건 / 서버 {', '.join(servers)}"


class LogPipeline:
    """조립·흐름만 담당. 구체 저장소를 직접 만들지 않고 주입받는다."""
    def __init__(self, parser: LogParser, storage: LogStorage, formatter: SummaryFormatter):
        self.parser, self.storage, self.formatter = parser, storage, formatter
        self.entries = []

    def collect(self, raw_text):
        self.entries.extend(self.parser.parse(raw_text))

    def errors(self):
        return [e for e in self.entries if e[0] == "ERROR"]

    def save(self):
        self.storage.write(f"{server}: {message}" for _, server, message in self.errors())

    def summary(self):
        return self.formatter.format(self.errors())


RAW = """
INFO|web-1|요청 처리 완료
ERROR|db-2|연결 시간 초과
ERROR|web-1|응답 코드 500
WARN|web-2|느린 응답
ERROR|db-2|디스크 부족
"""

if __name__ == "__main__":
    # 1) 메모리 저장소로 파일 없이 검증
    mem = MemoryLogStorage()
    p = LogPipeline(LogParser(), mem, SummaryFormatter())
    p.collect(RAW)
    p.save()
    assert p.summary() == "오류 3건 / 서버 db-2, web-1"
    assert mem.lines == ["db-2: 연결 시간 초과", "web-1: 응답 코드 500", "db-2: 디스크 부족"]

    # 2) 파일 저장소로 바꿔도 LogPipeline은 그대로 — 시작 코드와 같은 파일 내용
    out = os.path.join(tempfile.gettempdir(), "q2_errors_solution.txt")
    p2 = LogPipeline(LogParser(), FileLogStorage(out), SummaryFormatter())
    p2.collect(RAW)

def contract_violations(skill):

class Berserk(Skill):
    """mp가 충분할 때만 발동, 대신 피해 2배. 약속 ①②③ 모두 지킴."""

            return 0
        hero.mp -= self.cost
        return self.power * 2

    def apply(self, hero):
        if hero.mp < self.cost:
            return 0
        hero.mp -= self.cost
        hero.hp += self.amount
        return self.amount


class Drain(RecoverySkill):

항목배점확인 방법
공개 테스트 t1·t2(기존 세 수단)3t1 통과 1 + t2 통과 2(t2 안의 두 assert 중 하나라도 실패하면 t2 0)

채점용 추가 케이스(학생 비공개)

  • pay(10000, PointPayment()) → "포인트 결제 10000원 + 수수료 100원 = 10100원" (1%가 정확히 최소값인 경계)
  • pay(10099, PointPayment()) → 수수료 100원(버림) / pay(10100, PointPayment()) → 수수료 101원
  • pay(0, CardPayment()) → 수수료 0원 / pay(49, CardPayment()) → 수수료 0원 / pay(1, BankTransfer()) → 수수료 500원
  • label을 @property로 준 새 수단도 동작해야 함 / fee를 빼먹은 하위 클래스는 만들 때 TypeError

허용 해법: label은 클래스 속성·@property·__init__에서 고정값 대입 모두 인정. typing.Protocol로 만든 경우 t3(isabstract)이 실패하므로 지시 위반으로 3점 중 1점.

"""Q4 정답 코드(교사용): 결제 수단을 추상화해 OCP를 지키는 구조 (Python 3.12+)

공개 테스트(문제와 동일) 5개 + 채점용 추가 케이스(answers.html에만 공개)를 assert로 확인한다.

실행: python3 q4_payment.py   -> 마지막 줄 "Q4 정답 코드: 모든 assert 통과"
"""
from abc import ABC, abstractmethod


class PaymentMethod(ABC):
    """결제 수단의 약속: label(표시 이름, str)과 fee(amount) -> 0 이상의 int 수수료."""
    label: str = ""

    @abstractmethod
    def fee(self, amount: int) -> int: ...


class CardPayment(PaymentMethod):
    label = "카드"
    def fee(self, amount):
        return amount * 2 // 100


class BankTransfer(PaymentMethod):
    label = "계좌이체"
    def fee(self, amount):
        return 500


class PhonePayment(PaymentMethod):
    label = "휴대폰"
    def fee(self, amount):
        return amount * 5 // 100


class PointPayment(PaymentMethod):          # (2) 새 수단: 기존 클래스는 한 줄도 고치지 않고 추가만
    label = "포인트"
    def fee(self, amount):
        return max(100, amount // 100)      # 결제액의 1%(원 미만 버림), 최소 100원


class PayCounter:
    """결제 수단의 구체 타입을 모른다. 추상화(PaymentMethod)의 약속만 사용."""
    def pay(self, amount: int, method: PaymentMethod) -> str:
        fee = method.fee(amount)
        return f"{method.label} 결제 {amount}원 + 수수료 {fee}원 = {amount + fee}원"


# ===================== 공개 테스트 (수정 금지) =====================
def _check(name, fn):
    try:
        fn()
        print(f"[통과] {name}")
        return True
    except Exception as e:  # 실패해도 다음 테스트를 계속 실행
        print(f"[실패] {name} -> {type(e).__name__}: {e}")
        return False

"""Q5 정답 코드(교사용): 역할 인터페이스 분리(ISP) + 경로 계획기 주입(DIP) (Python 3.12+)

공개 테스트(문제와 동일) 4개 + 채점용 추가 케이스(answers.html에만 공개)를 assert로 확인한다.

실행: python3 q5_delivery_robot.py   -> 마지막 줄 "Q5 정답 코드: 모든 assert 통과"
"""
from abc import ABC, abstractmethod


# ---- ISP: 쓰는 쪽(Dispatcher)이 필요로 하는 역할 단위로 쪼갠 인터페이스 ----
class Courier(ABC):
    """모든 배달 로봇의 공통 역할: 이동하고 보관함을 연다."""
    def __init__(self):
        self.log = []

    @abstractmethod
    def drive_to(self, place): ...

    @abstractmethod
    def open_locker(self): ...


class ElevatorUser(ABC):
    """승강기를 탈 수 있는 로봇만 갖는 역할."""
    @abstractmethod
    def ride_elevator(self, floor): ...


class SidewalkRobot(Courier):                      # 못 하는 일은 아예 약속하지 않는다
    def drive_to(self, place):
        self.log.append(f"이동:{place}")

    def open_locker(self):
        self.log.append("보관함 열림")


class BuildingRobot(Courier, ElevatorUser):        # 두 역할을 모두 구현
    def drive_to(self, place):
        self.log.append(f"이동:{place}")

    def open_locker(self):
        self.log.append("보관함 열림")

    def ride_elevator(self, floor):
        self.log.append(f"승강기:{floor}층")


# ---- DIP: 상위 정책(Dispatcher)과 하위 세부(GpsMap)가 모두 추상화에 의존 ----
class RoutePlanner(ABC):
    @abstractmethod
    def route(self, start, goal) -> list[str]: ...


class GpsMap(RoutePlanner):
    def route(self, start, goal):
        return [start, "중앙광장", goal]


class Dispatcher:
    def __init__(self, planner: RoutePlanner):     # 구체 지도 클래스를 직접 만들지 않고 주입받는다
        self.planner = planner

    def deliver(self, robot: Courier, start, goal, floor=None):
        if floor is not None and not isinstance(robot, ElevatorUser):   # 역할(추상) 확인은 허용
            raise ValueError("이 로봇은 승강기를 탈 수 없어 층 배달을 맡을 수 없다")
        for place in self.planner.route(start, goal):
            robot.drive_to(place)
        if floor is not None:
            robot.ride_elevator(floor)
        robot.open_locker()
        return robot.log


# ===================== 공개 테스트 (수정 금지) =====================
def _check(name, fn):
    try:
        fn()
        print(f"[통과] {name}")
        return True
    except Exception as e:  # 실패해도 다음 테스트를 계속 실행
        print(f"[실패] {name} -> {type(e).__name__}: {e}")
        return False


def t1_roles():
    assert issubclass(SidewalkRobot, Courier) and issubclass(BuildingRobot, Courier)
    assert issubclass(BuildingRobot, ElevatorUser)
    assert not issubclass(SidewalkRobot, ElevatorUser), "인도 로봇이 승강기 역할을 떠안고 있음"
    assert not hasattr(SidewalkRobot(), "ride_elevator"), "인도 로봇에 쓸 수 없는 메서드가 남아 있음"


def t2_injected_planner():
    class FakePlanner(RoutePlanner):            # 테스트용 가짜 경로 계획기
        def route(self, start, goal):
            return [start, goal]
    log = Dispatcher(FakePlanner()).deliver(SidewalkRobot(), "정문", "도서관 앞")
    assert log == ["이동:정문", "이동:도서관 앞", "보관함 열림"], log


def t3_building_with_floor():
    log = Dispatcher(GpsMap()).deliver(BuildingRobot(), "기숙사", "공학관", floor=4)
    assert log == ["이동:기숙사", "이동:중앙광장", "이동:공학관", "승강기:4층", "보관함 열림"], log


def t4_refuse_before_moving():
    robot = SidewalkRobot()
    try:
        Dispatcher(GpsMap()).deliver(robot, "정문", "공학관", floor=3)
    except ValueError:
        assert robot.log == [], "거절하기 전에 이미 움직였음"
        return
    raise AssertionError("승강기를 못 타는 로봇에 층 배달을 맡겼는데 ValueError가 없음")


if __name__ == "__main__":
    print("현재 동작:", Dispatcher(GpsMap()).deliver(BuildingRobot(), "기숙사", "공학관", floor=4))
    tests = [("역할 인터페이스 분리", t1_roles), ("경로 계획기 주입", t2_injected_planner),
             ("건물 로봇 층 배달", t3_building_with_floor), ("못 하는 일은 출발 전 거절", t4_refuse_before_moving)]
    passed = sum(_check(n, f) for n, f in tests)
    print(f"공개 테스트 {passed}/{len(tests)} 통과")
    assert passed == len(tests)

    # ---- 채점용 추가 케이스(학생 비공개) ----
    import inspect
    # 1) 테스트가 새로 만든 로봇 종류(드론형 건물 로봇)도 Dispatcher 수정 없이 동작해야 함

        def __init__(self):
            self.log = []                          # log를 각 로봇에서 준비하는 허용 해법에서도 동작
        def drive_to(self, place): self.log.append(f"비행:{place}")
        def open_locker(self): self.log.append("투하")
        def ride_elevator(self, floor): self.log.append(f"옥상→{floor}층")
    class OnePlanner(RoutePlanner):
        def route(self, start, goal): return [goal]
    assert Dispatcher(OnePlanner()).deliver(RoofDrone(), "a", "b", floor=2) == ["비행:b", "옥상→2층", "투하"]
    # 2) 층 없는 배달은 건물 로봇도 승강기를 타지 않음
    assert Dispatcher(OnePlanner()).deliver(BuildingRobot(), "a", "b") == ["이동:b", "보관함 열림"]
    # 3) floor=0(지하/로비 등)도 '층 지정'으로 취급(falsy 함정)
    assert Dispatcher(OnePlanner()).deliver(BuildingRobot(), "a", "b", floor=0) == ["이동:b", "승강기:0층", "보관함 열림"]
    # 4) 역할 인터페이스는 추상이어야 함
    assert inspect.isabstract(Courier) and inspect.isabstract(ElevatorUser) and inspect.isabstract(RoutePlanner)
    # 5) 우회 오답 탐지: Dispatcher가 구체 클래스 이름을 알면 안 됨
    disp_src = inspect.getsource(Dispatcher)
    for banned in ("GpsMap", "SidewalkRobot", "BuildingRobot", "hasattr", "NotImplementedError"):
        assert banned not in disp_src, banned
    print("Q5 정답 코드: 모든 assert 통과")
