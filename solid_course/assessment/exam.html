"""Q2 시작 코드: 서버 로그 수집기 (Python 3.12+)

여러 서버가 보낸 로그 문자열을 모아, 오류만 골라 파일에 저장하고
운영 회의용 요약 문장을 만든다. 이 코드는 정상 동작한다. 구조를 분석하라.

실행: python3 q2_log_collector.py
"""
import os
import tempfile


class ServerLogCollector:
    def __init__(self, path):                                   # 1
        self.path = path                                        # 2
        self.entries = []                                       # 3
                                                                # 4
    def collect(self, raw_text):                                # 5
        # 로그 한 줄 형식: "레벨|서버이름|메시지"                  # 6
        for row in raw_text.strip().splitlines():               # 7
            level, server, message = row.split("|")             # 8
            self.entries.append((level.strip(), server.strip(), message.strip()))  # 9
                                                                # 10
    def errors(self):                                           # 11
        return [e for e in self.entries if e[0] == "ERROR"]     # 12
                                                                # 13
    def save(self):                                             # 14
        with open(self.path, "w", encoding="utf-8") as f:       # 15
            for level, server, message in self.errors():        # 16
                f.write(f"{server}: {message}\n")               # 17
                                                                # 18
    def summary(self):                                          # 19
        errs = self.errors()                                    # 20
        servers = sorted({server for _, server, _ in errs})     # 21
        return f"오류 {len(errs)}건 / 서버 {', '.join(servers)}"  # 22


if __name__ == "__main__":
    raw = """
    INFO|web-1|요청 처리 완료
    ERROR|db-2|연결 시간 초과
    ERROR|web-1|응답 코드 500
    WARN|web-2|느린 응답
    ERROR|db-2|디스크 부족
    """
    out = os.path.join(tempfile.gettempdir(), "q2_errors.txt")
    c = ServerLogCollector(out)
    c.collect(raw)
    c.save()
    print(c.summary())
    with open(out, encoding="utf-8") as f:
        print(f.read(), end="")

상황: 게임의 자동 전투 함수 auto_battle은 Skill 문서에 적힌 약속 ①②③만 믿고 작성되었습니다. 여러 개발자가 각자 새 스킬을 하위 클래스로 추가했습니다.
"""Q3 시작 코드: 게임 캐릭터 스킬 (Python 3.12+)

전투 시스템(auto_battle)은 Skill의 약속만 믿고 작성되었다.
실행 전에 출력을 먼저 예측하라.

실행: python3 q3_game_skills.py
"""


class Hero:
    def __init__(self, name, hp, mp):
        self.name, self.hp, self.mp = name, hp, mp


class Skill:
    """Skill의 약속(모든 스킬이 지켜야 함)
    ① 사용 전: mp가 cost보다 적으면 아무것도 바꾸지 않고 0을 돌려준다.
    ② 사용 후: mp를 정확히 cost만큼 줄이고, 0 이상의 피해량(int)을 돌려준다.
    ③ hp는 스킬 사용으로 바뀌지 않는다.
    """
    def __init__(self, name, cost, power):
        self.name, self.cost, self.power = name, cost, power

    def use(self, hero):
        if hero.mp < self.cost:
            return 0
        hero.mp -= self.cost
        return self.power


class Fireball(Skill):
    """mp가 20 이상 남아 있으면 피해 +5"""
    def use(self, hero):
        bonus = 5 if hero.mp >= 20 else 0
        damage = super().use(hero)
        return damage + bonus if damage > 0 else 0


class Berserk(Skill):
    """mp가 모자라면 hp를 대신 써서라도 발동한다"""
    def use(self, hero):
        if hero.mp < self.cost:
            hero.hp -= self.cost
            return self.power * 2
        return super().use(hero)


class Drain(Skill):
    """적의 생명력을 빼앗아 자신을 회복한다(피해량을 음수로 표시)"""
    def use(self, hero):
        hero.mp -= self.cost
        hero.hp += self.power
        return -self.power


def auto_battle(hero, skills, boss_hp):
    for skill in skills:
        damage = skill.use(hero)
        boss_hp -= damage
        print(f"{skill.name:<4} 피해 {damage:>3} | 보스 {boss_hp:>3} | {hero.name} hp {hero.hp} mp {hero.mp}")
    return boss_hp


if __name__ == "__main__":
    hero = Hero("린", hp=40, mp=25)
    skills = [Fireball("화염구", 10, 30), Drain("흡수", 8, 15),
              Berserk("광폭", 12, 20), Skill("베기", 0, 10)]
    left = auto_battle(hero, skills, boss_hp=100)
    print("남은 보스 hp:", left)

상황: 결제 창구 PayCounter는 결제 수단별 수수료를 붙여 영수증 한 줄을 만듭니다. 결제 수단이 문자열이라 새 수단이 생길 때마다 pay() 안의 if-elif를 고쳐야 합니다.
"""Q4 시작 코드: 결제 수수료와 영수증 한 줄 (Python 3.12+)

지금은 결제 수단이 문자열이고, 새 수단이 생길 때마다 PayCounter.pay 안의
if-elif를 고쳐야 한다. 문제의 요구대로 리팩터링하라.
맨 아래 공개 테스트는 수정하지 말고, 모두 [통과]가 나오게 만들어라.

실행: python3 q4_payment.py
"""


class PayCounter:
    def pay(self, amount, method):
        if method == "card":
            label, fee = "카드", amount * 2 // 100          # 결제액의 2%(원 미만 버림)
        elif method == "bank":
            label, fee = "계좌이체", 500                    # 고정 500원
        elif method == "phone":
            label, fee = "휴대폰", amount * 5 // 100        # 결제액의 5%(원 미만 버림)
        else:
            raise ValueError(f"모르는 결제 수단: {method}")
        return f"{label} 결제 {amount}원 + 수수료 {fee}원 = {amount + fee}원"


# ===================== 공개 테스트 (수정 금지) =====================
def _check(name, fn):
    try:
        fn()
        print(f"[통과] {name}")
        return True
    except Exception as e:  # 실패해도 다음 테스트를 계속 실행
        print(f"[실패] {name} -> {type(e).__name__}: {e}")
        return False


def t1_card():
    assert PayCounter().pay(12345, CardPayment()) == "카드 결제 12345원 + 수수료 246원 = 12591원"


def t2_bank_phone():
    assert PayCounter().pay(3000, BankTransfer()) == "계좌이체 결제 3000원 + 수수료 500원 = 3500원"
    assert PayCounter().pay(999, PhonePayment()) == "휴대폰 결제 999원 + 수수료 49원 = 1048원"


def t3_abstract():
    import inspect
    assert inspect.isabstract(PaymentMethod), "PaymentMethod는 추상 클래스여야 함"
    for cls in (CardPayment, BankTransfer, PhonePayment):
        assert issubclass(cls, PaymentMethod), f"{cls.__name__}가 PaymentMethod를 상속하지 않음"


def t4_point():
    assert PayCounter().pay(5000, PointPayment()) == "포인트 결제 5000원 + 수수료 100원 = 5100원"
    assert PayCounter().pay(25000, PointPayment()) == "포인트 결제 25000원 + 수수료 250원 = 25250원"


def t5_new_method_without_edit():
    # 테스트가 즉석에서 만든 새 결제 수단. PayCounter를 고치지 않고도 동작해야 한다.
    class GiftCard(PaymentMethod):
        label = "상품권"
        def fee(self, amount):
            return 0
    assert PayCounter().pay(7000, GiftCard()) == "상품권 결제 7000원 + 수수료 0원 = 7000원"
